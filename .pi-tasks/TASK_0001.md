```sh
# Capture pre-audit state (all files except audit.md)
cd /data/data/com.termux/files/home/arinanoX
find . -not -name audit.md -not -path './.git/*' -type f -exec sha256sum {} \; | sort > /tmp/pre_audit_checksums.txt

# Run the audit — generate audit.md
scripts=$(jq -r '.scripts // empty | keys[]' package.json 2>/dev/null || true)
if [ -z "$scripts" ]; then
  # Edge case: no scripts key
  echo "No scripts found in package.json" > audit.md
else
  echo "# npm Script Audit" > audit.md
  for script in $scripts; do
    echo "## Script: $script" >> audit.md
    exit_code=0
    output=$(npm run "$script" 2>&1) || exit_code=$?
    echo "Exit code: $exit_code" >> audit.md
    echo "Output:" >> audit.md
    echo '```' >> audit.md
    echo "$output" >> audit.md
    echo '```' >> audit.md
    if [ "$exit_code" -ne 0 ]; then
      echo "Root cause: Script \"$script\" failed with exit code $exit_code." >> audit.md
    else
      echo "Root cause: N/A — script completed successfully." >> audit.md
    fi
    echo "" >> audit.md
  done
fi

# Verify audit.md was created
test -f audit.md || { echo "FAIL: audit.md not created"; exit 1; }

# Verify audit.md documents the scripts state correctly
if [ -z "$scripts" ]; then
  grep -qi "no scripts" audit.md && echo "PASS: no-scripts case documented" || { echo "FAIL: missing no-scripts statement"; exit 1; }
else
  for script in $scripts; do
    grep -q "## Script: $script" audit.md || { echo "FAIL: script $script not in audit.md"; exit 1; }
    grep -q "Exit code:" audit.md || { echo "FAIL: exit code missing for $script"; exit 1; }
    grep -q "Root cause:" audit.md || { echo "FAIL: root cause missing for $script"; exit 1; }
  done
  echo "PASS: all scripts documented in audit.md"
fi

# Verify no other files mutated
find . -not -name audit.md -not -path './.git/*' -type f -exec sha256sum {} \; | sort > /tmp/post_audit_checksums.txt
diff /tmp/pre_audit_checksums.txt /tmp/post_audit_checksums.txt && echo "PASS: project tree unchanged (except audit.md)" || { echo "FAIL: project tree changed"; exit 1; }
```
